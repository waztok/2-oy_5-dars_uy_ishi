#include <stdio.h>

int main() {
    int a[4][4] = {
        {1, 0, 3, 0},
        {0, 5, 6, 7},
        {8, 9, 0, 2},
        {0, 4, 0, 1}
    };

    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            if (a[i][j] == 0) {
                a[i][j] = 1;
            }
        }
    }

    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            printf("%d ", a[i][j]);
        }
        printf("\n");
    }

    return 0;
}
