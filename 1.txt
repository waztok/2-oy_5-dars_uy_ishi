#include <stdio.h>

int main() {
    int a[3][4] = {
        {3, 8, 2, 5},
        {9, 1, 7, 4},
        {6, 12, 5, 10}
    };

    for (int i = 0; i < 3; i++) {
        int max = a[i][0];
        for (int j = 1; j < 4; j++) {
            if (a[i][j] > max) {
                max = a[i][j];
            }
        }
        printf("%d-qator eng kattasi: %d\n", i + 1, max);
    }

    return 0;
}
