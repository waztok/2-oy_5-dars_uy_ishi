#include <stdio.h>

int main() {
    int a[4][4] = {
        {12, 5, 8, 20},
        {3, 15, 1, 9},
        {11, 6, 25, 4},
        {7, 2, 14, 10}
    };

    int max = a[0][0];
    int min = a[0][0];

    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            if (a[i][j] > max) {
                max = a[i][j];
            }
            if (a[i][j] < min) {
                min = a[i][j];
            }
        }
    }

    printf("Eng katta: %d\n", max);
    printf("Eng kichik: %d\n", min);
    printf("Ayirmasi: %d\n", max - min);

    return 0;
}
